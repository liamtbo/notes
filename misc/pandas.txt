# pandas view
"""When you create a view of a DataFrame in Pandas, it creates a new object 
    where the data attribute references the original data in memory. That means 
    any changes made in the view reflect in the original DataFrame. """
    
import pandas as pd
# Sample DataFrame
df = pd.DataFrame({'A': [1, 2, 3], 'B': [4, 5, 6]})
# Creating a view
view_df = df[['A']]
view_df['A'][0] = 99
print(df)

# pandas copy
"""In contrast, when you create a copy of a DataFrame, it gets a separate instance in 
memory. Any modifications made to the copy will not affect the original DataFrame."""

copy_df = df.copy()
copy_df['A'][0] = 99
print(df)
print(copy_df)
